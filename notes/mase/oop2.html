1. Stream Gatherers

1.1. Example on gatherer

var windows = Stream.of(1, 2, 3, 4, 5)
        .gather(Gatherers.windowSliding(3)).toList();

System.out.println(windows);
// [[1, 2, 3], [2, 3, 4], [3, 4, 5]]

1.2. Common Gatherers

  • fold: stateful reduction that continues the stream
  • mapConcurrent: concurrent mapping
  • scan: running accumulation
  • windowFixed: non-overlapping batches
  • windowSliding: sliding windows

2. Stream Basics

2.1. Sum

import java.util.*;
import java.util.stream.*;

List<Integer> numbers = List.of(5, 2, 9, 1, 7);

int sum = numbers.stream()
        .mapToInt(Integer::intValue)
        .sum();

// same but using reduce
int sum = numbers.stream()
        .reduce(0, Integer::sum);

System.out.println(sum); // 24

2.2. Average

double avg = numbers.stream()
        .mapToInt(Integer::intValue)
        .average()
        .orElse(0);

System.out.println(avg); // 4.8

2.3. Sorting

// ascending
List<Integer> sorted = numbers.stream()
        .sorted()
        .toList();

// descending
List<Integer> sortedDesc = numbers.stream()
        .sorted(Comparator.reverseOrder())
        .toList();

2.4. Filtering

List<Integer> evens = numbers.stream()
        .filter(n -> n % 2 == 0)
        .toList();

System.out.println(evens); // [2]

2.5. Map

List<Integer> squared = numbers.stream()
        .map(n -> n * n)
        .toList();

System.out.println(squared); // [25, 4, 81, 1, 49]

2.6. Min/Max

int max = numbers.stream()
        .max(Integer::compareTo)
        .orElseThrow();

int min = numbers.stream()
        .min(Integer::compareTo)
        .orElseThrow();

2.7. Count

long count = numbers.stream()
        .filter(n -> n > 3)
        .count();

2.8. Distinct

List<Integer> distinct = Stream.of(1,2,2,3,3,3)
        .distinct()
        .toList();

2.9. Limit/Skip

List<Integer> first3 = numbers.stream()
        .limit(3)
        .toList();

List<Integer> skip2 = numbers.stream()
        .skip(2)
        .toList();

2.10. Collect to Map

record Person(String name, int age) {}

List<Person> people = List.of(
        new Person("Alice", 30),
        new Person("Bob", 25)
);

Map<String, Integer> map = people.stream()
        .collect(Collectors.toMap(
                Person::name,
                Person::age
        ));

2.11. Grouping

Map<Integer, List<Person>> byAge = people.stream()
        .collect(Collectors.groupingBy(Person::age));

2.12. Any / All / None Match

boolean anyEven = numbers.stream()
        .anyMatch(n -> n % 2 == 0);

boolean allPositive = numbers.stream()
        .allMatch(n -> n > 0);

boolean noneNegative = numbers.stream()
        .noneMatch(n -> n < 0);

2.13. Joining Stings

List<String> words = List.of("Java", "Streams", "API");

String result = words.stream()
        .collect(Collectors.joining(", "));

2.14. Peek (debugging)

numbers.stream()
        .filter(n -> n > 3)
        .peek(System.out::println)
        .toList();

3. Map to Map basics

3.1. Stream into a Map

List<String> names = List.of("Alice", "Bob", "Charlie");

Map<String, Integer> nameLengthMap =
    names.stream()
    .collect(Collectors.toMap(
                              name -> name,
                              name -> name.length()
                              ));

System.out.println(nameLengthMap);
// {Bob=3, Alice=5, Charlie=7}

3.2. Object to Map

List<Person> people = List.of(
                              new Person("1", "Alice", 30),
                              new Person("2", "Bob", 25)
                              );

Map<String, Integer> ageMap = people.stream()
    .collect(Collectors.toMap(
                              Person::getName,
                              Person::getAge
                              ));

System.out.println(ageMap);
// {Bob=25, Alice=30}

3.3. Duplicate Keys

List<String> names = List.of("Alice", "Bob", "Alice");

Map<String, Integer> map =
    names.stream()
         .collect(Collectors.toMap(
             name -> name,
             name -> 1,
             (existing, replacement) -> existing + replacement
         ));

System.out.println(map);
// {Alice=2, Bob=1}

3.4. Map to Map

Map<String, Integer> original = Map.of(
        "Alice", 30,
        "Bob", 25
);

Map<String, String> transformed =
        original.entrySet()
                .stream()
                .collect(Collectors.toMap(
                        entry -> entry.getKey().toUpperCase(),
                        entry -> "Age: " + entry.getValue()
                ));

System.out.println(transformed);
// {ALICE=Age: 30, BOB=Age: 25}

3.5. Grouping into a Map 1 (by first letter)

List<String> names = List.of("Alice", "Bob", "Charlie", "Anna");

Map<Character, List<String>> grouped =
        names.stream()
                .collect(Collectors.groupingBy(
                        name -> name.charAt(0)
                ));

System.out.println(grouped);
// {A=[Alice, Anna], B=[Bob], C=[Charlie]}

3.6. Grouping into a Map 2 (has starting letter "A")

List<String> names = List.of("Alice", "Bob", "Charlie", "Anna");

Map<Boolean, List<String>> grouped =
        names.stream()
                .collect(Collectors.groupingBy(
                        name -> name.startsWith("A")
                ));

System.out.println(grouped);
// {false=[Bob, Charlie], true=[Alice, Anna]}

3.7. Mapping Values Only (Keep Keys)

Map<String, Integer> scores = Map.of(
    "Alice", 80,
    "Bob", 90
);

Map<String, Integer> boosted =
    scores.entrySet()
          .stream()
          .collect(Collectors.toMap(
              Map.Entry::getKey,
              entry -> entry.getValue() + 10
          ));

System.out.println(boosted);
// {Alice=90, Bob=100}

4. Records

4.1. Example

public record Person(String name, int age) {}
Person p = new Person("Alice", 30);
System.out.println(p.name)
System.out.println(p.age)

4.2. Validate inside of the constructor

public record Person(String name, int age) {

    public Person {
        if (age < 0) {
            throw new IllegalArgumentException("Age cannot be negative");
        }
    }
}

4.3. Custom Methods

public record Rectangle(double width, double height) {

    public double area() {
        return width * height;
    }
}

4.4. Static Fields and Methods

public record MathConstants(double value) {

    public static final double PI = 3.14159;

    public static MathConstants of(double v) {
        return new MathConstants(v);
    }
}

4.5. Implementing Interfaces

public record User(String username) implements Comparable<User> {

    @Override
    public int compareTo(User other) {
        return this.username.compareTo(other.username);
    }
}

5. Generics

5.1. Common Naming Conventions

  • T: Type
  • E: Element
  • K: Key
  • V: Value
  • R: Return type

5.2. Class example

class Box<T> {
    private T value;

    public void set(T value) {
        this.value = value;
    }

    public T get() {
        return value;
    }
}

5.3. Method example

public static <T> void printArray(T[] arr) {
    for (T item : arr) {
        System.out.println(item);
    }
}

// usage
Integer[] nums = {1,2,3};
String[] words = {"a","b"};

printArray(nums);
printArray(words);

5.4. Bounded Generics

Upper Bound

<T extends Number>

public static double sum(List<? extends Number> numbers) {
    double total = 0;
    for (Number n : numbers) {
        total += n.doubleValue();
    }
    return total;
}

// usage
List<Integer> ints = List.of(1,2,3);
List<Double> doubles = List.of(1.2, 2.3);

sum(ints);
sum(doubles);

Lower Bound

  • Use super instead of extends

5.5. Generics in Lambdas

Function<T,R>  // T → R
Predicate<T>   // T → boolean
Consumer<T>    // T → void
Supplier<T>    // () → T

6. Scoped Values

6.1. Basic Example Of Scoped Values

import java.lang.ScopedValue;

public class ScopedValueExample {

    private static final ScopedValue<String> USER = ScopedValue.newInstance();

    public static void main(String[] args) {

        ScopedValue.where(USER, "Alice").run(() -> {
            greet();
        });

        // Outside scope – would throw exception
        // System.out.println(USER.get());
    }

    static void greet() {
        System.out.println("Hello " + USER.get());
    }
}

6.2. Nested Scoped Values

private static final ScopedValue<String> USER = ScopedValue.newInstance();

public static void main(String[] args) {

    ScopedValue.where(USER, "Alice").run(() -> {

        System.out.println(USER.get()); // Alice

        ScopedValue.where(USER, "Bob").run(() -> {
            System.out.println(USER.get()); // Bob
        });

        System.out.println(USER.get()); // Alice again
    });
}

6.3. Using Scoped Values with Virtual Threads

import java.lang.ScopedValue;

public class VirtualThreadExample {

    private static final ScopedValue<String> REQUEST_ID = ScopedValue.newInstance();

    public static void main(String[] args) {

        ScopedValue.where(REQUEST_ID, "REQ-123").run(() -> {

            Thread.startVirtualThread(() -> {
                System.out.println("Request: " + REQUEST_ID.get());
            });
        });
    }
}

6.4. Scoped Values with Structured Concurrency

import java.lang.ScopedValue;
import java.util.concurrent.StructuredTaskScope;

public class StructuredExample {

    private static final ScopedValue<String> USER = ScopedValue.newInstance();

    public static void main(String[] args) throws Exception {

        ScopedValue.where(USER, "Alice").run(() -> {

            try (var scope = new StructuredTaskScope.ShutdownOnFailure()) {

                scope.fork(() -> {
                    return "Task1 sees " + USER.get();
                });

                scope.fork(() -> {
                    return "Task2 sees " + USER.get();
                });

                scope.join();
                scope.throwIfFailed();
            }
        });
    }
}

6.5. Using call() Instead of run()

String result = ScopedValue.where(USER, "Alice")
        .call(() -> "Hello " + USER.get());

System.out.println(result);

7. Flexible Constructor Bodies

7.1. Concept

  • It is now possible to:
    • Have a super() call after some validation is done on the params passed through the constructor and the same applies to a call to this()
    • Can also log with System.out before a call to super or this
    • Not allowed: access to instance attributes (members)

8. Compact Source Files

8.1. Allow behaviour like in a modern language, no need for a class around main

void main() {
    System.out.println("Hello, world!");
}

8.2. Something like this is also now allowed

int base = 10;

void main() {
    greet("Alice");
}

void greet(String name) {
    System.out.println(name + " has base " + base);
}
